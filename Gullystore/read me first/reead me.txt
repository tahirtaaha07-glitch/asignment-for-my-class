import os, json
from fastapi import FastAPI, Request, Form
from fastapi.responses import HTMLResponse, RedirectResponse
from fastapi.templating import Jinja2Templates
from starlette.middleware.sessions import SessionMiddleware
from datetime import datetime

app = FastAPI()
app.add_middleware(SessionMiddleware, secret_key="gully_stealth_2026_pro")
templates = Jinja2Templates(directory="templates")

DATA_FILE = "data.json"
ORDERS_FILE = "orders.json"

# --- FIXED DATABASE FUNCTION WITH AUTO-1000 ITEMS ---
def get_db(file=DATA_FILE):
    if not os.path.exists(file) or (file == DATA_FILE and os.path.getsize(file) < 50):
        if file == DATA_FILE:
            # Generate 1,000 Items Automatically
            products = []
            tech_imgs = [
                "https://images.unsplash.com/photo-1591799264318-7e6ef8ddb7ea?w=500", 
                "https://images.unsplash.com/photo-1587202372775-e229f172b9d7?w=500",
                "https://images.unsplash.com/photo-1542751371-adc38448a05e?w=500"
            ]
            brands = ["ASUS", "MSI", "Intel", "AMD", "Nvidia"]
            parts = ["GPU", "CPU", "RAM", "SSD", "Board"]
            
            for i in range(1, 1001):
                products.append({
                    "id": i,
                    "name": f"{brands[i%5]} {parts[i%5]} Pro-v{i}",
                    "specs": f"High-performance {parts[i%5]} for gaming. Version {i}.",
                    "price": 5000 + (i * 100),
                    "image": tech_imgs[i%3]
                })
            
            default = {"users": {"admin": "1234"}, "products": products}
        else:
            default = []
            
        with open(file, "w") as f: 
            json.dump(default, f, indent=4)
        return default
    
    with open(file, "r") as f: 
        return json.load(f)

def save_db(data, file=DATA_FILE):
    with open(file, "w") as f: 
        json.dump(data, f, indent=4)

@app.get("/", response_class=HTMLResponse)
async def home(request: Request, q: str = None):
    db = get_db()
    products = db.get("products", [])
    if q:
        products = [p for p in products if q.lower() in p["name"].lower()]
    # Note: Using "items" here to match your home.html loop
    return templates.TemplateResponse("home.html", {"request": request, "items": products, "user": request.session.get("user")})

@app.post("/add_product")
async def add_product(name: str = Form(...), specs: str = Form(...), price: int = Form(...), image: str = Form(...)):
    db = get_db()
    new_id = max([p["id"] for p in db["products"]], default=0) + 1
    db["products"].append({"id": new_id, "name": name, "specs": specs, "price": price, "image": image})
    save_db(db)
    return RedirectResponse(url="/admin", status_code=303)

@app.post("/update_product/{item_id}")
async def update_product(item_id: int, name: str = Form(...), specs: str = Form(...), price: int = Form(...), image: str = Form(...)):
    db = get_db()
    for p in db["products"]:
        if p["id"] == item_id:
            p.update({"name": name, "specs": specs, "price": price, "image": image})
    save_db(db)
    return RedirectResponse(url="/admin", status_code=303)

@app.get("/delete_product/{item_id}")
async def delete_product(item_id: int):
    db = get_db()
    db["products"] = [p for p in db["products"] if p["id"] != item_id]
    save_db(db)
    return RedirectResponse(url="/admin", status_code=303)

@app.post("/confirm_order")
async def confirm_order(request: Request, p_name: str = Form(...), p_price: int = Form(...)):
    user = request.session.get("user")
    user_name = user["name"] if user else "Customer"
    orders = get_db(ORDERS_FILE)
    order_id = f"GULLY-{datetime.now().strftime('%M%S')}"
    
    orders.append({"id": order_id, "user": user_name, "item": p_name, "price": p_price, "time": datetime.now().strftime("%Y-%m-%d %H:%M")})
    save_db(orders, ORDERS_FILE)
    
    my_num = "923327606015"
    msg = f"Salam! I saw this on your Store! I want to buy {p_name} (Rs. {p_price:,}). Order ID: {order_id}"
    return RedirectResponse(url=f"https://wa.me/{my_num}?text={msg.replace(' ', '%20')}", status_code=303)

@app.get("/admin", response_class=HTMLResponse)
async def admin(request: Request):
    user = request.session.get("user")
    if not user or user.get("type") != "admin": return RedirectResponse(url="/login")
    db = get_db()
    orders = get_db(ORDERS_FILE)
    total_rev = sum(int(o['price']) for o in orders)
    return templates.TemplateResponse("admin.html", {"request": request, "items": db["products"], "orders": orders, "revenue": total_rev})

@app.get("/login", response_class=HTMLResponse)
async def login_page(request: Request):
    return templates.TemplateResponse("login.html", {"request": request})

@app.post("/login")
async def handle_login(request: Request, username: str = Form(...), password: str = Form(...)):
    db = get_db()
    if username.lower() == "admin" and password == db["users"]["admin"]:
        request.session["user"] = {"name": "Admin", "type": "admin", "pic": "https://ui-avatars.com/api/?name=A&background=ea3a0a&color=fff"}
        return RedirectResponse(url="/admin", status_code=303)
    return RedirectResponse(url="/login")

@app.post("/auth/google")
async def google_auth(request: Request, credential: str = Form(...)):
    request.session["user"] = {"name": "Customer", "type": "google", "pic": "https://ui-avatars.com/api/?name=U&background=ea3a0a&color=fff"}
    return RedirectResponse(url="/", status_code=303)

@app.get("/logout")
async def logout(request: Request):
    request.session.clear()
    return RedirectResponse(url="/", status_code=303)
